    <script>
      const T = %%TL%%;
      window.__timelines = window.__timelines || {};
      const tl = gsap.timeline({ paused: true });
      const ME = document.getElementById("me");

      // no speaker: the glows drift slowly for the whole reel
      if (document.getElementById("bgfx")) {
        tl.fromTo("#glow1", { x: 0, y: 0, scale: 1 }, { x: 160, y: 120, scale: 1.12, duration: T.dur, ease: "sine.inOut" }, 0);
        tl.fromTo("#glow2", { x: 0, y: 0, scale: 1.1 }, { x: -180, y: -140, scale: 1, duration: T.dur, ease: "sine.inOut" }, 0);
      }

      // captions: the thin row eases in, heavy words land one by one with the speech
      T.caps.forEach((c, i) => {
        const cap = document.getElementById("cap" + i);
        cap.querySelectorAll(".w").forEach(w => {
          const t = +w.dataset.t;
          if (w.classList.contains("thin"))
            tl.fromTo(w, { opacity: 0, y: 18, filter: "blur(6px)" }, { opacity: 1, y: 0, filter: "blur(0px)", duration: 0.3, ease: "power2.out" }, t);
          else
            tl.fromTo(w, { opacity: 0, scale: 1.2 }, { opacity: 1, scale: 1, duration: 0.22, ease: "power3.out" }, t);
        });
        tl.to(cap, { opacity: 0, y: -20, duration: 0.18, ease: "power1.in" }, c.end - 0.18);
      });

      // big framing for the hook and the call to action; switch it under a cutaway so nobody sees the jump
      if (ME) {
        tl.set("#me", { transformOrigin: `540px ${T.bigOrigin}px` }, 0);
        T.big.forEach(([a, b]) => {
          tl.set("#me", { y: T.bigY, scale: T.bigScale }, a);
          if (b !== null) tl.set("#me", { y: 0, scale: 1 }, b);
        });

        // joins between takes: the dissolve is baked in by join.py, a short blur hides what is left
        tl.set("#mev", { scale: 1.04 }, 0);
        T.joins.forEach(t => {
          tl.fromTo("#mev", { filter: "blur(0px)" }, { filter: "blur(9px)", duration: 0.14, ease: "power1.in", immediateRender: false }, t - 0.06);
          tl.to("#mev", { filter: "blur(0px)", duration: 0.22, ease: "power1.out" }, t + 0.08);
        });
      }

      // cutaways: soft crossfade; a card followed straight away by another stays underneath until covered
      T.broll.forEach(([id, a, b], k) => {
        const next = T.broll[k + 1], covered = next && next[1] - b < 0.05;
        tl.fromTo("#" + id, { opacity: 0 }, { opacity: 1, duration: 0.4, ease: "sine.inOut", immediateRender: false }, a - 0.2);
        tl.fromTo(`#${id} .m`, { scale: 1 }, { scale: 1.06, duration: b - a + 0.6, ease: "none" }, a - 0.2);
        if (covered) tl.set("#" + id, { opacity: 0 }, b + 0.25);
        else tl.to("#" + id, { opacity: 0, duration: 0.4, ease: "sine.inOut" }, b - 0.05);
      });

      if (T.cta) {
        // the arrow's tip lands just past the button's right edge, whatever the length of the text
        const box = document.querySelector("#cta span"), arrow = document.getElementById("arrow");
        if (box && arrow) arrow.style.left = Math.min(70 + box.offsetWidth - 6, 1080 - 400) + "px";
        tl.fromTo("#cta", { opacity: 0, y: 30, scale: 0.9 }, { opacity: 1, y: 0, scale: 1, duration: 0.4, ease: "back.out(1.8)" }, T.cta.at);
        if (T.cta.arrow) {
          tl.to(["#ushaft", "#ashaft"], { strokeDashoffset: 0, duration: 0.95, ease: "power1.inOut" }, T.cta.at + 0.35);
          tl.to(["#uhead", "#ahead"], { strokeDashoffset: 0, duration: 0.22, ease: "power1.out" }, T.cta.at + 1.25);
        }
        tl.to(["#cta", "#arrow"], { opacity: 0, duration: 0.5, ease: "sine.inOut" }, T.cta.until);
      }
      if (T.end && ME) tl.to("#me", { opacity: 0, duration: 0.7, ease: "sine.inOut" }, T.end);
      if (T.logo !== null) tl.fromTo("#logo", { opacity: 0, scale: 0.94 }, { opacity: 1, scale: 1, duration: 0.8, ease: "sine.out" }, T.logo);

      // the reel's own idea (custom.js): one effect that illustrates a word
